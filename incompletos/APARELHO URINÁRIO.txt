    var limitacoes = [];

    function laudoLimitacoes() {
        limitacoes = [];
        const meteorismo = $('#checkLimMet').is(':checked');
        const biotipo = $('#checkLimBio').is(':checked');
        const posicionamento = $('#checkLimPos').is(':checked');

        if (meteorismo) {
            limitacoes.push('por distensão gasosa intestinal');
        }
        if (biotipo) {
            limitacoes.push('pela atenuação do feixe acústico inerente ao biotipo');
        }
        if (posicionamento) {
            limitacoes.push('devido restrições de posicionamento e cooperação');
        }

        let frase = laudoAparelho();
        if (limitacoes.length > 0) {
            if (frase == "") {
                frase = "Exame limitado parcialmente ";
            } else {
                frase = frase.slice(0,-9) + ", limitado parcialmente ";
            }
            if (limitacoes.length == 1) {
                frase += limitacoes[0];
            } else if (limitacoes.length == 2) {
                frase += limitacoes[0] + " e " + limitacoes[1];
            } else {
                frase += limitacoes[0] + ", " + limitacoes[1] + " e " + limitacoes[2];
            }
            frase += ".<br><br>";
        }

        return frase;
    }

    function adicionarCistoRenal() {
        const formulario = {
            tipo: $('input[name="checkCistoRenalTipo[]"]:checked').val(),
            composicao: $('input[name="checkCistoRenalComposicao[]"]:checked').val(),
            topografia: $('input[name="checkCistoRenalTopografia[]"]:checked').val(),
            localizacao: $('#inputCisRenLoc option:checked').text(),
            lateralidade: $('input[name="checkCistoRenalLateralidade[]"]:checked').val(),
            medida1: tratarFloatModal('CisRenMed1'),
            medida2: tratarFloatModal('CisRenMed2'),
            medida3: tratarFloatModal('CisRenMed3'),
        };

        if (formulario.medida1 == null) {
            $('#alertaCistoRenal').html('Necessário incluir ao menos 1 medida.').show();
            $('#inputCistoRenalMedida1').focus();
            return false;
        }

        lesoesRenais.push({
            id: uuidv4(),
            lesao: "cisto",
            tipo: formulario.tipo,
            composicao: formulario.composicao,
            topografia: formulario.topografia,
            localizacao: formulario.localizacao,
            lateralidade: formulario.lateralidade,
            medida1: formulario.medida1,
            medida2: formulario.medida2,
            medida3: formulario.medida3,
        });
        listarCistosRenais();

        $('#modalCistoRenal').modal('hide');
    }

    function listarCistosRenais() {
        $('#div-rins-cistos').html('');
        for (var i = 0; i < lesoesRenais.length; i++) {
            var texto = '';
            if (lesoesRenais[i].lesao == 'cisto') {
                if (lesoesRenais[i].tipo == 'unico') {
                    var topografia = (lesoesRenais[i].topografia == 'seio' ? 'do seio renal' : 'cortical');
                    if (lesoesRenais[i].medida2 == null && lesoesRenais[i].medida3 == null) {
                        texto = `cisto ${lesoesRenais[i].composicao} em ${lesoesRenais[i].localizacao} ${topografia} à ${lesoesRenais[i].lateralidade}, medindo cerca de ${lesoesRenais[i].medida1} cm`;
                    } else if (lesoesRenais[i].medida3 == null) {
                        texto = `cisto ${lesoesRenais[i].composicao} em ${lesoesRenais[i].localizacao} ${topografia} à ${lesoesRenais[i].lateralidade}, medindo cerca de ${lesoesRenais[i].medida1} x ${lesoesRenais[i].medida2} cm`;
                    } else {
                        texto = `cisto ${lesoesRenais[i].composicao} em ${lesoesRenais[i].localizacao} ${topografia} à ${lesoesRenais[i].lateralidade}, medindo cerca de ${lesoesRenais[i].medida1} x ${lesoesRenais[i].medida2} x ${lesoesRenais[i].medida3} cm`;
                    }
                } else {
                    var topografia = (lesoesRenais[i].topografia == 'seio' ? 'em seio renal' : 'corticais');
                    if (lesoesRenais[i].medida2 == null && lesoesRenais[i].medida3 == null) {
                        texto = `múltiplos cistos ${lesoesRenais[i].composicao} ${topografia}, o maior localizado no ${lesoesRenais[i].localizacao} à ${lesoesRenais[i].lateralidade}, medindo cerca de ${lesoesRenais[i].medida1} cm`;
                    } else if (lesoesRenais[i].medida3 == null) {
                        texto = `múltiplos cistos ${lesoesRenais[i].composicao} ${topografia}, o miaor localizado no ${lesoesRenais[i].localizacao} à ${lesoesRenais[i].lateralidade}, medindo cerca de ${lesoesRenais[i].medida1} x ${lesoesRenais[i].medida2} cm`;
                    } else {
                        texto = `múltiplos cistos ${lesoesRenais[i].composicao} ${localizacao}, o maior localizado no ${lesoesRenais[i].localizacao} à ${lesoesRenais[i].lateralidade}, medindo cerca de ${lesoesRenais[i].medida1} x ${lesoesRenais[i].medida2} x ${lesoesRenais[i].medida3} cm`;
                    }
                }
                $('#div-rins-cistos').append('<div class="lesao-focal rounded fs--2 mb-1 px-2 py-1">' + texto + `<div class="lesao-focal-lixeira" onclick="deletarCistoRenal('${lesoesRenais[i].id}');"><i class="fa-fw fa-solid fa-close"></i></div></div>`);
            }
        }
        montarLaudo(CKEDITOR.instances['editor']);
    }

    function deletarCistoRenal(id) {
        lesoesRenais = lesoesRenais.filter(function(item) {
            return item.id !== id
        })
        listarCistosRenais();
    }
