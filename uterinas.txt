function evaluarSGAvsFGR() {

	var UADopplerPI = get_UADopplerPI()
	var MCADopplerMCAPI = get_MCADopplerMCAPI()
	var CPRPercentile = get_CPRPercentile()
	var DVPIPercentile = get_DVPIPercentile()
	var valorGA = get_GA()

	var UADiastolicFlow = $('#select_diastolic_flow :selected').text()
	var DVAtrialFlow = $('#FGR_vs_SGA_select_atrial_flow :selected').text()

	$("#Note_fetalGrowth1").text('')
	$("#Note_fetalGrowth").text('')

	hide_panels_below_ctg()
	hide_panels_all_ctg()
	clearResults()

	if (UADiastolicFlow == 'Reverse' && (DVAtrialFlow == 'Not-available' || DVAtrialFlow == 'Present')) {
		// CTG 1 - DONE
		console.info('c1')
		panel_CTG()
		return
	}

	if (UADiastolicFlow == "Absent" && valorGA >= 30 && valorGA < 34
	 	&& DVAtrialFlow == "Present" && DVPIPercentile > 95) {

		setResultText('Stage 3 *', 'Delivery +')
		$('#persistentText').show()
		$('#steroidsText').show()

		console.info('r8')
		return
	}

	if ((UADiastolicFlow == "Absent" || UADopplerPI > 95) && (DVAtrialFlow == "Present" && DVPIPercentile > 95)) {

		if (valorGA >= 30) {
			setResultText('Stage 3 *', 'Delivery +')
			$('#persistentText').show()
			$('#steroidsText').show()

			if (UADiastolicFlow == "Absent") {
				console.info('r8')
			} else if (UADopplerPI > 95) {
				console.info('r13')
			}

			return
		}

		if (UADiastolicFlow == "Absent") {
			// CTG 2 - DONE
			console.info('c1')
			panel_CTG()
			return
		}

		if (UADopplerPI > 95) {
			// CTG 4 --> CTG panel variant 1 - DONE
			console.info('c3')
			panel_CTG_variant1()
			return
		}
	}

	if (UADiastolicFlow == "Absent" && (DVAtrialFlow == "Not-available" || DVPIPercentile < 95)) {
		// CTG 3 - DONE
		console.info('c1')
		panel_CTG()
		return
	}

	if (UADopplerPI > 95 && (DVPIPercentile < 95 || DVAtrialFlow == "Not-available")) {
		// CTG 5 --> CTG panel variant 1 - DONE
		console.info('c3')
		panel_CTG_variant1()
		return
	}

	if ((MCADopplerMCAPI < 5 || CPRPercentile < 5) && (DVPIPercentile < 95 || DVAtrialFlow == 'Not-available')) {
		// CTG - 7 --> CTG panel variant 3 - DONE
		panel_CTG_variant3()
		console.info('c5')
		return
	}

	if ((MCADopplerMCAPI < 5 || CPRPercentile < 5) && DVPIPercentile > 95) {
		if (valorGA >= 34) {
			setResultText('Stage 3', 'Delivery after 30')
			console.info('r19')
			return
		}

		// CTG - 6 --> CTG panel variant 2 - DONE
		panel_CTG('variant2')
		console.info('c4')
		return
	}
}
